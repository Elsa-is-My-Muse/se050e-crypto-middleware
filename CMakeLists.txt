cmake_minimum_required(VERSION 3.19)
project(se050e_crypto_middleware LANGUAGES C)

# Source inventory is shared with the copy/export and validation tools.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/manifest.json" _se050_manifest)
function(se050_read_list output key absolute_paths)
    string(JSON _count LENGTH "${_se050_manifest}" "${key}")
    set(_items)
    if(_count GREATER 0)
        math(EXPR _last "${_count} - 1")
        foreach(_index RANGE 0 ${_last})
            string(JSON _item GET "${_se050_manifest}" "${key}" ${_index})
            if(absolute_paths)
                list(APPEND _items "${CMAKE_CURRENT_SOURCE_DIR}/${_item}")
            else()
                list(APPEND _items "${_item}")
            endif()
        endforeach()
    endif()
    set(${output} "${_items}" PARENT_SCOPE)
endfunction()

# No implicit MCU/RTOS selection: an empty or fake production port is an error.
set(SE050_OS_PORT_HEADER "" CACHE STRING "Selected port header, for example se050_port_rtthread.h")
set(SE050_PORT_INCLUDE_DIRS "" CACHE STRING "OS/HAL/board include directories (absolute paths)")
set(SE050_PLATFORM_SOURCES "" CACHE STRING "Exactly one OS/board implementation (absolute source paths)")
option(SE050_WITH_MBEDTLS "Build the optional PSA/Mbed TLS adapter" OFF)
set(SE050_MBEDTLS_INCLUDE_DIRS "" CACHE STRING "Selected Mbed TLS/TF-PSA public and configuration headers")
set(SE050_MBEDTLS_LIBRARIES "" CACHE STRING "Parent-provided Mbed TLS/TF-PSA targets/libraries")
set(SE050_MBEDTLS_COMPILE_DEFINITIONS "" CACHE STRING "Configuration macros matching the selected TLS libraries")

# Normalize Windows command-line paths before CMake creates source targets.
foreach(_paths SE050_PORT_INCLUDE_DIRS SE050_PLATFORM_SOURCES SE050_MBEDTLS_INCLUDE_DIRS)
    string(REPLACE "\\" "/" ${_paths} "${${_paths}}")
endforeach()

if(NOT SE050_OS_PORT_HEADER OR NOT SE050_PORT_INCLUDE_DIRS OR NOT SE050_PLATFORM_SOURCES)
    message(FATAL_ERROR "Select a real OS header, include paths and board implementation; see README.md")
endif()
foreach(_source IN LISTS SE050_PLATFORM_SOURCES)
    if(NOT IS_ABSOLUTE "${_source}" OR NOT EXISTS "${_source}")
        message(FATAL_ERROR "SE050_PLATFORM_SOURCES requires existing absolute paths: ${_source}")
    endif()
endforeach()

se050_read_list(_core core_sources TRUE)
se050_read_list(_includes core_include_dirs TRUE)
se050_read_list(_defines compile_definitions FALSE)
add_library(se050_core STATIC ${_core} ${SE050_PLATFORM_SOURCES})
add_library(SE050::Core ALIAS se050_core)
target_compile_features(se050_core PUBLIC c_std_99)
target_include_directories(se050_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" ${SE050_PORT_INCLUDE_DIRS})
target_include_directories(se050_core PRIVATE ${_includes})
target_compile_definitions(se050_core PUBLIC "SE050_PORT_HEADER=\"${SE050_OS_PORT_HEADER}\"")
target_compile_definitions(se050_core PRIVATE ${_defines})

if(SE050_WITH_MBEDTLS)
    if(NOT SE050_MBEDTLS_INCLUDE_DIRS)
        message(FATAL_ERROR "SE050_WITH_MBEDTLS needs the matching TLS/TF-PSA include paths")
    endif()
    se050_read_list(_middleware middleware_sources TRUE)
    add_library(se050_mbedtls STATIC ${_middleware})
    add_library(SE050::MbedTLS ALIAS se050_mbedtls)
    target_compile_features(se050_mbedtls PUBLIC c_std_99)
    target_include_directories(se050_mbedtls PUBLIC ${SE050_MBEDTLS_INCLUDE_DIRS})
    target_compile_definitions(se050_mbedtls PUBLIC ${SE050_MBEDTLS_COMPILE_DEFINITIONS})
    target_link_libraries(se050_mbedtls PUBLIC SE050::Core ${SE050_MBEDTLS_LIBRARIES})
endif()
